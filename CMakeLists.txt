cmake_minimum_required(VERSION 3.16)
project(3DSLink LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(3dslink-host source/Server.cpp source/Host.cpp)
target_include_directories(3dslink-host PRIVATE include)
target_compile_options(3dslink-host PRIVATE -Wall -Wextra -Wpedantic)
option(ENABLE_SANITIZERS "Enable address and undefined behavior checks" OFF)
if(ENABLE_SANITIZERS)
    target_compile_options(3dslink-host PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(3dslink-host PRIVATE -fsanitize=address,undefined)
endif()
